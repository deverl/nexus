#!/usr/bin/env bash

set -euo pipefail

usage() {
    cat << 'EOF'

Usage: ssh.pod [OPTIONS] [NAMESPACE] [PATTERN]

Exec into the first running pod whose name matches the given pattern in the specified namespace.

Arguments:
  NAMESPACE    Kubernetes namespace (required)
               Example: nrac, trustpoint, homewi, acclhw, ...

  PATTERN      String/regex pattern to match pod name against (required)
               Example: core, beat, celery, rabbitmq, warranty, affiliate, portal

Options:
  -d, --dev    Use the dev KUBECONFIG (~/.kube/dev_config)
  -h, --help   Show this help message

Examples:
  ssh.pod nrac core
  ssh.pod trustpoint warranty
  ssh.pod --dev nrac core

EOF
    exit 1
}

DEV=0

# Parse options
while [[ $# -gt 0 ]]; do
    case "$1" in
        -d|--dev)
            DEV=1
            shift
            ;;
        -h|--help)
            usage
            ;;
        --)
            shift
            break
            ;;
        -*)
            printf "ERROR: Unknown option '%s'\n" "$1" >&2
            usage
            ;;
        *)
            break
            ;;
    esac
done

if [[ $# -ne 2 ]]; then
    usage
fi

NAMESPACE="$1"
PATTERN="$2"

# Use the dev KUBECONFIG if requested
if [[ "$DEV" -eq 1 ]]; then
    export KUBECONFIG=~/.kube/dev_config
fi

# Find the first running pod whose name matches the pattern
pod=$(kubectl get pods -n "$NAMESPACE" --field-selector=status.phase=Running \
    -o jsonpath='{range .items[*]}{.metadata.name}{"\n"}{end}' \
    | grep -E "$PATTERN" | head -n 1)

if [[ -z "$pod" ]]; then
    printf "ERROR: No running pod found matching pattern '%s' in namespace '%s'\n" \
        "$PATTERN" "$NAMESPACE" >&2
    exit 1
fi

printf "Found pod: %s\n" "$pod"
printf "Entering shell...\n\n"

# Exec into the pod
kubectl exec -it -n "$NAMESPACE" "$pod" -- /bin/bash || {
    # Fallback to sh if bash is not available
    echo "bash not found, trying /bin/sh..."
    kubectl exec -it -n "$NAMESPACE" "$pod" -- /bin/sh
}
